       ```python
       class Session:
           def __init__(self, user_id: str):
               self.session_id = str(uuid4())
               self.user_id = user_id
               self.buffer = SlidingWindowBuffer(window_size=30)
               self.classifier = MockLSTM()

               self.phrase_buffer = []  # Accumulate recognized signs

       class SessionManager:
           def __init__(self):
               self._sessions: Dict[str, Session] = {}  # sid -> Session
               self._user_sessions: Dict[str, str] = {}  # user_id -> sid

           def create_session(self, sid: str, user_id: str) -> Session:
               if user_id in self._user_sessions:
                   old_sid = self._user_sessions[user_id]
                   self._sessions.pop(old_sid, None)
               session = Session(user_id)
               self._sessions[sid] = session
               self._user_sessions[user_id] = sid
               return session

           def get_session(self, sid: str) -> Optional[Session]:
               return self._sessions.get(sid)

           def remove_session(self, sid: str):
               session = self._sessions.pop(sid, None)
               if session:
                   self._user_sessions.pop(session.user_id, None)

           def clear_phrase(self, sid: str):
               session = self.get_session(sid)
               if session:
                   session.phrase_buffer.clear()

       session_manager = SessionManager()
       ```

       ```python
       from typing import Optional, Dict, Any
       from app.core.sliding_window import SlidingWindowBuffer
       from app.models.mock_lstm import MockLSTM
       from app.schemas.landmarks import LandmarksPayload

       class RecognitionPipeline:
           def __init__(self, window_size: int = 30):
               self.buffer = SlidingWindowBuffer(window_size=window_size)
               self.classifier = MockLSTM()
               self._last_sign_time = 0
               self._phrase_timeout_ms = 2500  # D-05

           def process_landmarks(self, landmarks_data: LandmarksPayload) -> Optional[Dict[str, Any]]:
               # Convert Pydantic model to numpy array
               import numpy as np
               frame = self._extract_features(landmarks_data)

               # Add to buffer
               window = self.buffer.add_landmarks(frame)

               if window is None:
                   return None  # Buffer not full yet

               # Run classification
               result = self.classifier.predict(window)
               self._last_sign_time = landmarks_data.timestamp

               return {
                   'sign': result['sign'],
                   'confidence': result['confidence'],
                   'timestamp': landmarks_data.timestamp
               }

           def check_phrase_complete(self, current_time: int) -> bool:
               if self._last_sign_time == 0:
                   return False
               return (current_time - self._last_sign_time) > self._phrase_timeout_ms

           def get_phrase(self) -> list:
               phrase = self.buffer.get_accumulated_signs()
               self.buffer.clear()
               self._last_sign_time = 0
               return phrase

           def _extract_features(self, payload: LandmarksPayload) -> np.ndarray:
               """Extract 126-dimensional feature vector from landmarks."""
               import numpy as np
               features = np.zeros(126, dtype=np.float32)

               if 'left' in payload.landmarks and payload.landmarks['left']:
                   left_data = payload.landmarks['left']
                   for i, lm in enumerate(left_data.landmarks):
                       base = i * 3
                       features[base] = lm.x
                       features[base + 1] = lm.y
                       features[base + 2] = lm.z

               if 'right' in payload.landmarks and payload.landmarks['right']:
                   right_data = payload.landmarks['right']
                   for i, lm in enumerate(right_data.landmarks):
                       base = 63 + i * 3  # Start at index 63
                       features[base] = lm.x
                       features[base + 1] = lm.y
                       features[base + 2] = lm.z

               return features
       ```

       ```python
       import os
       from jose import jwt, JWTError
       from socketio import AsyncServer

       SECRET_KEY = os.getenv('JWT_SECRET', 'dev-secret-change-in-production')
       ALGORITHM = 'HS256'

       async def verify_token(token: str) -> Optional[str]:
           try:
               payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
               return payload.get('sub') or payload.get('userId') or payload.get('user_id')
           except JWTError:
               return None
       ```

    2. `recognition-service/app/api/socket_handlers.py`:
       ```python
       from socketio import AsyncServer
       from app.services.session_manager import session_manager
       from app.core.recognition_pipeline import RecognitionPipeline
       from app.schemas.landmarks import LandmarksPayload
       from app.middleware.auth import verify_token
       import json

       sio = AsyncServer(async_mode='asgi', cors_allowed_origins='*')
       pipeline = RecognitionPipeline(window_size=30)

       @sio.event
       async def connect(sid: str, environ: dict, auth: dict):
           token = auth.get('token', '')
           user_id = await verify_token(token)
           if not user_id:
               await sio.disconnect(sid, reason='unauthenticated')
               return

           session = session_manager.create_session(sid, user_id)
           await sio.emit('connected', {'sessionId': session.session_id}, to=sid)
           print(f'User {user_id} connected (sid={sid})')

       @sio.event
       async def disconnect(sid: str):
           session = session_manager.get_session(sid)
           if session:
               print(f'User {session.user_id} disconnected')
           session_manager.remove_session(sid)

       @sio.event
       async def landmarks(sid: str, data: dict):
           session = session_manager.get_session(sid)
           if not session:
               await sio.emit('error', {'message': 'no_session'}, to=sid)
               return

           try:
               payload = LandmarksPayload(**data)
           except Exception as e:
               await sio.emit('error', {'message': f'invalid_payload: {str(e)}'}, to=sid)
               return

           result = session.classifier.process_landmarks(payload)
           if result:
               await sio.emit('sign_recognized', result, room=session.user_id)

           # Check for phrase completion
           if session.classifier.check_phrase_complete(payload.timestamp):
               phrase = session.classifier.get_phrase()
               await sio.emit('phrase_complete', {'text': ' '.join(s['sign'] for s in phrase)})
       ```
