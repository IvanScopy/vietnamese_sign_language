       ```python
       import subprocess

       @runtime_checkable
       class TTSProvider(Protocol):
           def synthesize(self, text: str, voice: str = None, speed: float = 1.0) -> bytes:

       class PiperTTS:
           """Piper TTS provider using piper CLI."""
           def __init__(self, voice: str = 'vivos'):
               self.voice = voice
               self.sample_rate = 22050

           def synthesize(self, text: str, voice: str = None, speed: float = 1.0) -> bytes:
               voice = voice or self.voice
               cmd = [
                   'piper', '--model', f'/usr/share/piper-voices/{voice}/model.onnx',
                   '--output_raw'
               ]
               proc = subprocess.run(
                   cmd,
                   input=text.encode('utf-8'),
                   capture_output=True,
                   timeout=5
               )
               if proc.returncode != 0:
                   raise TTSError(f'Piper failed: {proc.stderr.decode()}')
               # Convert raw PCM to WAV header
               return self._pcm_to_wav(proc.stdout, self.sample_rate)

           def _pcm_to_wav(self, pcm: bytes, sample_rate: int) -> bytes:
               import struct
               num_channels = 1
               bits_per_sample = 16
               byte_rate = sample_rate * num_channels * bits_per_sample // 8
               block_align = num_channels * bits_per_sample // 8
               data_size = len(pcm)
               file_size = 36 + data_size

               header = struct.pack('<4sI4s4sIHHIIHH4sI',
                 b'RIFF', file_size, b'WAVE',
                 b'fmt ', 16, 1, num_channels, sample_rate,
                 byte_rate, block_align, bits_per_sample * num_channels,
                 b'data', data_size
               )
               return header + pcm

       class EspeakTTS:
           """eSpeak-ng fallback provider."""
           def __init__(self, voice: str = 'vi'):
               self.voice = voice
               self.sample_rate = 22050

           def synthesize(self, text: str, voice: str = None, speed: float = 1.0) -> bytes:
               voice = voice or self.voice
               cmd = [
                   'espeak-ng', '-v', voice,
                   '--stdout', '--wav', text
               ]
               proc = subprocess.run(cmd, capture_output=True, timeout=5)
               if proc.returncode != 0:
                   raise TTSError(f'eSpeak failed: {proc.stderr.decode()}')
               return proc.stdout

       class TTSError(Exception):
           pass
       ```

    1. `recognition-service/app/services/tts_service.py`:
       ```python
       import os
       from typing import Optional
       from app.models.tts_provider import TTSProvider, PiperTTS, EspeakTTS, TTSError

       class TTSService:
           def __init__(self, primary: str = 'piper', fallback: str = 'espeak'):
               self.primary_provider = self._create_provider(primary)
               self.fallback_provider = self._create_provider(fallback) if primary != fallback else None

           def _create_provider(self, name: str) -> TTSProvider:
               if name == 'piper':
                   voice = os.getenv('PIPER_VOICE', 'vivos')
                   return PiperTTS(voice=voice)
               elif name == 'espeak':
                   voice = os.getenv('ESPEAK_VOICE', 'vi')
                   return EspeakTTS(voice=voice)
               else:
                   raise ValueError(f'Unknown TTS provider: {name}')

           def synthesize(self, text: str, voice: str = None, speed: float = 1.0) -> bytes:
               """Synthesize text to WAV bytes, with fallback on error."""
               try:
                   return self.primary_provider.synthesize(text, voice, speed)
               except TTSError as e:
                   if self.fallback_provider:
                       try:
                           return self.fallback_provider.synthesize(text, voice, speed)
                       except TTSError as fallback_e:
                           raise TTSError(f'Both providers failed: {e}, {fallback_e}')
                   raise

       tts_service = TTSService(
           primary=os.getenv('TTS_PROVIDER', 'piper'),
           fallback='espeak'
       )
       ```
